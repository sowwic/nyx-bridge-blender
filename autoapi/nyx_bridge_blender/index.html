    BlenderClient(*args, **kwargs)

    Bases: NyxClient

    Nyx client living in Blender.

    What Nyx sends is run on Blender's main thread, so it can use bpy safely. Scripts find bpy, bmesh and mathutils in their namespace.

    Serve from a background thread (start_in_thread, what start does) in a Blender with a window. In blender --background serve from the main thread with serve_forever: there is no event loop to hand work over to.

    def run_on_main_thread(self, func: typing.Callable[[], typing.Any]) -> typing.Any:
    """Run a function on Blender's main thread and wait for its result.

    Args:
        func (Callable[[], Any]): function to run.

    Returns:
        Any: whatever the function returns.
    """
    if threading.current_thread() is threading.main_thread():
        return func()
    return self._pump.submit(func)

    MainThreadPump()

    Runs functions on the main thread of Blender for threads that need it.

    Blender's Python API may only be used from the main thread. The pump lets other threads hand it a function and wait for the result: a timer of Blender, which runs on the main thread, picks the functions up (run_pending).

    def run_pending(self) -> None:
    """Run the functions that are waiting. Call it from the main thread."""
    while True:
        try:
            job = self._jobs.get_nowait()
        except queue.Empty:
            return
        if self._active:
            job.run()
        else:
            job.fail(RuntimeError("The client was stopped"))

    def start(self) -> None:
    """Start taking functions and ask Blender's timers to run them.

    Call it from the main thread.
    """
    import bpy

    self._active = True
    if not bpy.app.timers.is_registered(self._timer):
        bpy.app.timers.register(self._timer, first_interval=0.0, persistent=True)

    def submit(self, func: typing.Callable[[], typing.Any]) -> typing.Any:
    """Run a function on the main thread and wait for its result.

    Args:
        func (Callable[[], Any]): function to run.

    Raises:
        RuntimeError: the pump is not running.

    Returns:
        Any: whatever the function returns. What it raises is raised here.
    """
    if not self._active:
        raise RuntimeError("Nothing runs functions on the main thread of Blender")
    job = _Job(func)
    self._jobs.put(job)
    job.done.wait()
    if job.error is not None:
        raise job.error
    return job.result

    def serve_background(
    name: str | None = None,
    address: str | tuple[str, int] | None = None,
    token: str | None = None,
    connect_timeout: float | None = None,
) -> None:
    """Serve Nyx from the main thread of a Blender running with `--background`.

    There is no event loop to hand work to, so the requests run right on the calling
    thread. Returns when Nyx asks the client to shut down or the connection ends, it
    does not reconnect, and Blender then exits as its script is done. Start it from a
    Launch Process node of Nyx with a command like

        blender --background --python-expr "import nyx_bridge_blender;
        nyx_bridge_blender.serve_background()"

    The extension has to be installed and enabled for the package to be importable.

    Args:
        name (str | None): name Nyx shows for this Blender, `NYX_BRIDGE_CLIENT_NAME` if
            omitted, then derived from the process id.
        address (str | tuple[str, int] | None): `host:port` of Nyx, see `start`.
        token (str | None): token of the Nyx server.
        connect_timeout (float | None): seconds to keep trying to reach Nyx, forever if
            omitted.
    """
    BlenderClient(
        name=name or os.environ.get(CLIENT_NAME_VARIABLE) or None,
        address=address,
        token=token,
        reconnect=False,
    ).serve_forever(connect_timeout)

    def start(
    address: str | tuple[str, int] | None = None,
    token: str | None = None,
    name: str | None = None,
) -> BlenderClient:
    """Connect Blender to Nyx from a background thread, and keep trying until it works.

    Where Nyx is comes from the arguments, then the environment, then the address file
    Nyx writes, see `nyx_bridge.protocol.find_server_address`. It does not matter
    whether Nyx is already running: the client connects once it is, and again if
    the connection drops. Call it from the main thread.

    Args:
        address (str | tuple[str, int] | None): `host:port` of Nyx.
        token (str | None): token of the Nyx server.
        name (str | None): name Nyx shows for this Blender. Derived from the process id
            if omitted.

    Returns:
        BlenderClient: the running client. An already running one is returned as it
            is, the arguments are ignored.
    """
    global _client, _thread
    if is_running():
        return _client
    _client = BlenderClient(name=name, address=address, token=token)
    _thread = _client.start_in_thread()
    return _client

    def stop() -> None:
    """Disconnect from Nyx and stop the client started by `start`."""
    global _client, _thread
    client, thread = _client, _thread
    _client = _thread = None
    if client is not None:
        client.stop()
    if thread is not None and thread is not threading.current_thread():
        thread.join(timeout=5)
